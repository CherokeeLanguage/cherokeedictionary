    Verb getDisplayValueConjugationEngineV3(likespreadsheets) {
        def paramMap = [:]
        paramMap.habitual = likespreadsheets.vthirdpressylll
        paramMap.imperative = likespreadsheets.vsecondimpersylln
        paramMap.infinitive = likespreadsheets.vthirdinfsyllp
        paramMap.present1st = likespreadsheets.vfirstpresh
        paramMap.present3rd = likespreadsheets.syllabaryb
        paramMap.remotepast = likespreadsheets.vthirdpastsyllj
        paramMap.partofspeechc = likespreadsheets.partofspeechc
        paramMap.subject = PrefixTableSubject.SG1.toString()
        paramMap.object = PrefixTableObject.SG3AN.toString()
        paramMap.verbTense = Tense.PRESENT.toString()

        Verb displayValue
        try {
            def verb = VerbFactory.createVerbFromParameters(paramMap)
            verb = Conjugate.conjugate(verb);
            displayValue = verb
        } catch (Exception e) {
            e.printStackTrace()
            // ensure we never return a String here; caller will handle null
            displayValue = null
        }

        return displayValue
    }
